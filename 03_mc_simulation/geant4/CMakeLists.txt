cmake_minimum_required(VERSION 3.16)
project(graal_geant4 LANGUAGES CXX)
option(GRAAL_WITH_QT "Build the interactive Qt viewer as well as batch transport" OFF)
option(GRAAL_RESPONSE_ONLY "Build the offline ROOT diagnostic response without Geant4" OFF)
if(GRAAL_WITH_QT AND NOT GRAAL_RESPONSE_ONLY)
  find_package(Geant4 11.4 REQUIRED gdml qt vis_all)
elseif(NOT GRAAL_RESPONSE_ONLY)
  find_package(Geant4 11.4 REQUIRED gdml)
endif()
if(NOT GRAAL_RESPONSE_ONLY)
  include("${Geant4_USE_FILE}")
endif()
# Honour explicit ROOT_DIR/CMAKE_PREFIX_PATH; otherwise discover the active ROOT.
if(NOT ROOT_DIR AND NOT DEFINED ENV{ROOT_DIR})
  find_program(GRAAL_ROOT_CONFIG root-config)
  if(GRAAL_ROOT_CONFIG)
    execute_process(COMMAND "${GRAAL_ROOT_CONFIG}" --prefix
      OUTPUT_VARIABLE GRAAL_ROOT_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE
      RESULT_VARIABLE GRAAL_ROOT_CONFIG_STATUS)
    if(GRAAL_ROOT_CONFIG_STATUS EQUAL 0)
      list(APPEND CMAKE_PREFIX_PATH "${GRAAL_ROOT_PREFIX}")
    endif()
  endif()
endif()
find_package(ROOT 6.30 REQUIRED COMPONENTS Core RIO Tree TreePlayer)
include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(graal_mt_writer_check tests/check_ordered_transport.cc)
  target_compile_features(graal_mt_writer_check PRIVATE cxx_std_17)
  target_link_libraries(graal_mt_writer_check PRIVATE ROOT::Core ROOT::RIO ROOT::Tree Threads::Threads)
  add_executable(graal_compact_input_check tests/check_compact_input.cc)
  target_include_directories(graal_compact_input_check PRIVATE src)
  target_compile_features(graal_compact_input_check PRIVATE cxx_std_17)
  target_link_libraries(graal_compact_input_check PRIVATE ROOT::Core ROOT::RIO ROOT::Tree ROOT::TreePlayer)
endif()
add_executable(graal_digitize src/digitize.cc)
target_compile_features(graal_digitize PRIVATE cxx_std_17)
target_link_libraries(graal_digitize PRIVATE ROOT::Core ROOT::RIO ROOT::Tree ROOT::TreePlayer)
if(MSVC)
  target_compile_options(graal_digitize PRIVATE /W4)
else()
  target_compile_options(graal_digitize PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(graal_newplan src/newplan.cc)
target_compile_features(graal_newplan PRIVATE cxx_std_17)
target_link_libraries(graal_newplan PRIVATE ROOT::Core ROOT::RIO ROOT::Tree ROOT::TreePlayer)
if(MSVC)
  target_compile_options(graal_newplan PRIVATE /W4 /fp:strict)
else()
  target_compile_options(graal_newplan PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()
add_executable(graal_cylinders src/cylinders.cc)
target_compile_features(graal_cylinders PRIVATE cxx_std_17)
target_link_libraries(graal_cylinders PRIVATE ROOT::Core ROOT::RIO ROOT::Tree ROOT::TreePlayer)
if(MSVC)
  target_compile_options(graal_cylinders PRIVATE /W4 /fp:strict)
else()
  target_compile_options(graal_cylinders PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()
add_executable(graal_forward src/forward.cc)
target_compile_features(graal_forward PRIVATE cxx_std_17)
target_link_libraries(graal_forward PRIVATE ROOT::Core ROOT::RIO ROOT::Tree ROOT::TreePlayer)
if(MSVC)
  target_compile_options(graal_forward PRIVATE /W4 /fp:strict)
else()
  target_compile_options(graal_forward PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()
add_executable(graal_central src/central.cc)
target_compile_features(graal_central PRIVATE cxx_std_17)
target_link_libraries(graal_central PRIVATE ROOT::Core ROOT::RIO ROOT::Tree ROOT::TreePlayer)
if(MSVC)
  target_compile_options(graal_central PRIVATE /W4 /fp:strict)
else()
  target_compile_options(graal_central PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()
if(GRAAL_RESPONSE_ONLY)
  return()
endif()
add_executable(graal_geometry src/geometry.cc)
add_executable(graal_g4 src/transport.cc)
target_link_libraries(graal_g4 PRIVATE ROOT::Core ROOT::RIO ROOT::Tree ROOT::TreePlayer)
if(GRAAL_WITH_QT)
  target_compile_definitions(graal_g4 PRIVATE GRAAL_WITH_QT)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(graal_gdml_check tests/check_gdml.cc)
  add_executable(graal_abort_check tests/check_aborted_transport.cc)
  target_link_libraries(graal_abort_check PRIVATE ROOT::Core ROOT::RIO ROOT::Tree ROOT::TreePlayer)
  add_executable(graal_root_check tests/check_root_output.cc)
  target_compile_features(graal_root_check PRIVATE cxx_std_17)
  target_link_libraries(graal_root_check PRIVATE ROOT::Core ROOT::RIO ROOT::Tree)
endif()
set(graal_targets graal_geometry graal_g4)
if(BUILD_TESTING)
  list(APPEND graal_targets graal_gdml_check graal_abort_check)
endif()
foreach(target IN LISTS graal_targets)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE ${Geant4_INCLUDE_DIRS})
  target_link_libraries(${target} PRIVATE ${Geant4_LIBRARIES})
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
