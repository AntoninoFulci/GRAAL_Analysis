cmake_minimum_required(VERSION 3.16)
project(graal_geometry LANGUAGES CXX)
find_package(Geant4 11.4 REQUIRED gdml)
add_executable(graal_geometry src/geometry.cc)
target_compile_features(graal_geometry PRIVATE cxx_std_17)
target_include_directories(graal_geometry PRIVATE ${Geant4_INCLUDE_DIRS})
target_link_libraries(graal_geometry PRIVATE ${Geant4_LIBRARIES})
target_compile_options(graal_geometry PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(graal_gdml_check tests/check_gdml.cc)
  target_compile_features(graal_gdml_check PRIVATE cxx_std_17)
  target_include_directories(graal_gdml_check PRIVATE ${Geant4_INCLUDE_DIRS})
  target_link_libraries(graal_gdml_check PRIVATE ${Geant4_LIBRARIES})
endif()
